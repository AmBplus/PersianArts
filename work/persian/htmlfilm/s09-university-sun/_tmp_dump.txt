function render(n) {
  n = Math.max(0, Math.min(DURATION - 1, Math.round(n)));
  const T = n / FPS, b = T / BS;
  L.paper.cx = L.paper.origCx; L.paper.cy = L.paper.origCy; L.paper.w = L.paper.origW; L.paper.h = L.paper.origH;
  L.paper.markAt = 0; L.paper.stampAt = 99;
  ctx.setTransform(1, 0, 0, 1, 0, 0);
  // camera: a gentle designed move per ratio (keeps the composition inside the safe margin of both frames)
  const cz = WIDE ? 0.955 - 0.025 * ease.inOutSine(prog(b, 2, 12)) : 0.965 - 0.02 * ease.inOutSine(prog(b, 2, 13));
  const ccx = WIDE ? 960 + 34 * ease.inOutSine(prog(b, 8, 20)) : 540 + 22 * ease.inOutSine(prog(b, 6, 18));
  const ccy = WIDE ? 540 - 20 * ease.inOutSine(prog(b, 26, 40)) : 960 - 24 * ease.inOutSine(prog(b, 26, 42));
  ctx.setTransform(cz, 0, 0, cz, W / 2 - ccx * cz, H / 2 - ccy * cz);
  if (b < 5) scQuestion(b);
  else if (b < 10) scFaculty(b, 0);
  else if (b < 16) scFaculty(b, 1);
  else if (b < 19) scFaculty(b, 2);
  else if (b < 23) scFaculty(b, 3);
  else if (b < 35) scCampus(b);
  else scAnswer(b);
  // vignette + deterministic grain (screen space; keeps every frame textured; no wall-clock, no Math.random)
  ctx.setTransform(1, 0, 0, 1, 0, 0);
  const vg = ctx.createRadialGradient(W / 2, H / 2, Math.min(W, H) * 0.35, W / 2, H / 2, Math.max(W, H) * 0.75);
  vg.addColorStop(0, "rgba(0,0,0,0)"); vg.addColorStop(1, "rgba(0,0,0,.5)"); ctx.fillStyle = vg; ctx.fillRect(0, 0, W, H);
  ctx.save(); ctx.fillStyle = "#ffffff";
  for (let i = 0; i < grain.length; i++) { const g = grain[i], ox = ((g.x * W + n * (i % 7 + 1) * 3) % W), oy = ((g.y * H + n * (i % 5 + 1) * 2) % H);
    ctx.globalAlpha = g.a * 0.7; ctx.fillRect(ox, oy, g.r, g.r); } ctx.restore();
  ctx.setTransform(1, 0, 0, 1, 0, 0); ctx.globalAlpha = 1; ctx.direction = "ltr";   // leave no state for the next frame
  if (params.get("debug") === "2") { let d = document.getElementById("dbg"); if (!d) { d = document.createElement("pre"); d.id = "dbg"; d.style.cssText = "position:fixed;left:0;top:0;z-index:9;background:#fff;color:#000;font:20px monospace;padding:6px;max-width:1070px;white-space:pre-wrap"; document.body.appendChild(d); } d.textContent = "f" + n + " " + JSON.stringify(window.__cardDbg); }
  window.film.frame = n;
}
